• Use these three scratch-training commands. They differ only in label-context configuration.

  Important: do not use --classroom-overfit-all-races. Your training and validation database views must contain disjoint race IDs; training will stop if overlap is detected.

  ### Model A — current diffuse retrieval

  python train_model.py \
    --overwrite-existing \
    --output outputs/label_context_A_current.pt \
    --training-csv outputs/training_records_A.csv \
    --validation-csv outputs/validation_records_A.csv \
    --features-json tabfm_features.json \
    --epochs 30 \
    --auto-race-schedule \
    --query-races-per-step 5 \
    --learning-rate 0.0001 \
    --early-stopping-patience 10 \
    --fine-tune-scope full_model \
    --seed 42 \
    --device cpu \
    --race-context-mode self_attention \
    --race-context-dim 32 \
    --race-context-layers 1 \
    --race-context-heads 2 \
    --race-context-ff-dim 64 \
    --race-head-scale 0.1 \
    --encode-races-before-icl \
    --context-prototype-branch \
    --context-prototype-dim 16 \
    --context-prototype-max-correction 0.25 \
    --label-context-branch \
    --label-context-heads 2 \
    --label-context-max-correction 0.25 \
    --label-context-labels-in-values-only \
    --label-context-temperature 1.0 \
    --label-context-top-k 0 \
    --classification-loss-weight 1.0 \
    --pairwise-loss-weight 0.25 \
    --attention-delta-pairwise-loss-weight 0.05 \
    --context-prototype-loss-weight 0.25 \
    --label-context-loss-weight 0.25 \
    --context-dependence-loss-weight 0.10 \
    --context-dependence-margin 0.005 \
    --cardinality-loss-weight 0.05 \
    --checkpoint-metric composite \
    --max-grad-norm 5
